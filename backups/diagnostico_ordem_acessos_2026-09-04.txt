const TURNO_INICIO_A = 7;   // turno diurno começa às 07:00
const TURNO_INICIO_B = 19;  // turno noturno começa às 19:00
const TURNO_AVISO_MIN = 60; // mostrar card X minutos antes do fim

/* Retorna info do turno atual */
function getTurnoAtual(){
  const agora = new Date();
  const h = agora.getHours();
  const isDiurno = h >= TURNO_INICIO_A && h < TURNO_INICIO_B;
  const inicioH = isDiurno ? TURNO_INICIO_A : TURNO_INICIO_B;
  const fimH    = isDiurno ? TURNO_INICIO_B : TURNO_INICIO_A;
  // calcular início do turno
  const inicio = new Date(agora);
  if(isDiurno){
    inicio.setHours(TURNO_INICIO_A, 0, 0, 0);
  } else {
    if(h >= TURNO_INICIO_B){ inicio.setHours(TURNO_INICIO_B, 0, 0, 0); }
    else { inicio.setDate(inicio.getDate()-1); inicio.setHours(TURNO_INICIO_B, 0, 0, 0); }
  }
  // calcular fim
  const fim = new Date(inicio);
  fim.setHours(fim.getHours() + 12);
  const msParaFim = fim - agora;
  const minParaFim = Math.round(msParaFim / 60000);
  return {isDiurno, inicioH, fimH, inicio, fim, minParaFim,
    label: isDiurno ? 'Diurno 07:00 – 19:00' : 'Noturno 19:00 – 07:00'};
}

function getAcessosTurno(turno){
  const acc = state.accesses || [];
  return acc.filter(a => {
    const ts = new Date(a.createdAt || a.entrada || 0).getTime();
    return ts >= turno.inicio.getTime();
  });
}

function getAgendaTurno(){

}

  if(!remoteState||typeof remoteState!=='object')return localState;
  const merged={...localState};
  const remoteRole=remoteMeta?.role||remoteState?.meta?.updatedByPerfil||'';
  const remoteIsPortaria=String(remoteRole).toLowerCase()==='portaria';
  const localIsPortaria=!isAdmin();

  // Acessos: sempre mão dupla, com último update vencendo por registro.
  const deleted=mergeDeletedAccessIds(localState.deletedAccessIds,remoteState.deletedAccessIds);
  const deletedIds=new Set(deleted.map(x=>x.id));
  merged.deletedAccessIds=deleted;
  merged.accesses=mergeById(localState.accesses||[],remoteState.accesses||[],getAccessStamp,2500).filter(a=>!deletedIds.has(a.id));

  // Logs: todos podem gerar; admin visualiza completo.
  merged.logs=mergeById(localState.logs||[],remoteState.logs||[],x=>new Date(x?.criadoEm||0).getTime()||0,1500);

  // Módulos administrativos: portaria nunca sobrepõe. Snapshot de portaria não manda autoridade sobre moradores/admin/config.
  if(!remoteIsPortaria){
    merged.units=remoteState.units&&typeof remoteState.units==='object'?remoteState.units:(merged.units||{});
    merged.admin=remoteState.admin&&typeof remoteState.admin==='object'?remoteState.admin:merged.admin;
    merged.operational=Array.isArray(remoteState.operational)?remoteState.operational:(merged.operational||[]);
    merged.users=mergeUsersPreservingPasswords(localState?.users,Array.isArray(remoteState.users)&&remoteState.users.length?remoteState.users:(merged.users||defaultUsers()));
    merged.settings={...(merged.settings||{}),...(remoteState.settings||{})};
    merged.settings.theme=getLocalThemeMode();
    merged.passagens=mergeById(localState.passagens||[],remoteState.passagens||[],x=>new Date(x?.createdAt||x?.data||0).getTime()||0,500);
  }else if(localIsPortaria){
    // Mesmo quando o último snapshot foi feito pela portaria, ela continua recebendo esses módulos do estado remoto se existirem,
    // mas sem transformar isso em edição local; serve apenas para manter visualização consistente.
    merged.units=remoteState.units&&typeof remoteState.units==='object'?remoteState.units:(merged.units||{});
    merged.admin=remoteState.admin&&typeof remoteState.admin==='object'?remoteState.admin:merged.admin;
    merged.operational=Array.isArray(remoteState.operational)?remoteState.operational:(merged.operational||[]);
    merged.users=mergeUsersPreservingPasswords(localState?.users,Array.isArray(remoteState.users)&&remoteState.users.length?remoteState.users:(merged.users||defaultUsers()));
    merged.settings={...(merged.settings||{}),...(remoteState.settings||{})};
    merged.settings.theme=getLocalThemeMode();
  }
  merged.meta={...(merged.meta||{}),lastSyncMergeAt:new Date().toISOString()};
  return merged;
}

async function enterKioskMode(){
  document.body.classList.add('kiosk-mode');
  try{localStorage.setItem('pp_kiosk_mode','1')}catch(e){}
  try{
    if(document.documentElement.requestFullscreen && !document.fullscreenElement){
      await document.documentElement.requestFullscreen();
    }
  }catch(e){console.warn('Fullscreen não disponível ou bloqueado pelo navegador.',e)}
  updateKioskButton();
}
async function exitKioskMode(){
  document.body.classList.remove('kiosk-mode');
  try{localStorage.setItem('pp_kiosk_mode','0')}catch(e){}
  try{
    if(document.fullscreenElement && document.exitFullscreen){
      await document.exitFullscreen();
    }
  }catch(e){console.warn('Não foi possível sair do fullscreen pelo script.',e)}
  updateKioskButton();
}
function toggleKioskMode(){
  if(isKioskMode()) exitKioskMode();
  else enterKioskMode();
}

function reportPeriodRange(){
  const days=Number(document.getElementById('reportPeriod')?.value||30);
  if(!days)return {days:0,since:0,label:'Todo o histórico'};
  return {days,since:Date.now()-days*86400000,label:`Últimos ${days} dias`};
}
function parseDateMs(v){const d=new Date(v||0); return isNaN(d.getTime())?0:d.getTime()}
function accessMs(a){return parseDateMs(a?.entrada||((a?.data||'')+'T'+(a?.entradaHora||'00:00')))}
function opMs(o){return parseDateMs((o?.data||'')+'T00:00:00')}
function inReportRange(ms,since){return !since || ms>=since}
function topCounts(items,getKey,limit=5){
  const map=new Map();
  items.forEach(item=>{const key=(getKey(item)||'').trim(); if(!key||key==='—')return; map.set(key,(map.get(key)||0)+1)});
  return [...map.entries()].sort((a,b)=>b[1]-a[1]||a[0].localeCompare(b[0])).slice(0,limit);
}
function countResidents(){
  const units=Object.values(state.units||{});
  const people=units.flatMap(u=>Array.isArray(u.people)?u.people:[]).filter(p=>p&&p.nome);
  const app=people.filter(p=>String(p.app||'').toLowerCase()==='sim'||p.app===true).length;
  const bio=people.filter(p=>String(p.bio||'').toLowerCase()==='sim'||p.bio===true).length;
  const vehicles=units.flatMap(u=>Array.isArray(u.vehicles)?u.vehicles:[]).filter(v=>v&&(v.placa||v.modelo||v.cor||v.vaga));
  return {units,people,totalPeople:people.length,app,bio,vehicles:vehicles.length,occupied:units.filter(u=>(u.people||[]).some(p=>p.nome)).length};
}

function buildReportData(){

  const inside=(state.accesses||[]).filter(a=>!a.saida);
  const exits=accesses.filter(a=>a.saida);
  const ops=(state.operational||[]).filter(o=>inReportRange(opMs(o),range.since));

function agendaReportData(ops){
  const today=new Date().toISOString().slice(0,10);
  const now=new Date();
  const month=now.toISOString().slice(0,7);
  const year=String(now.getFullYear());
  const all=(ops||[]).filter(o=>o&&o.data);
  const inMonth=all.filter(o=>String(o.data||'').slice(0,7)===month);
  const inYear=all.filter(o=>String(o.data||'').slice(0,4)===year);
  const salaos=inMonth.filter(o=>o.tipo==='salao');
  const mudancas=inMonth.filter(o=>o.tipo==='mudanca');
  const salaosYear=inYear.filter(o=>o.tipo==='salao');
  const mudancasYear=inYear.filter(o=>o.tipo==='mudanca');
  const future=all.filter(o=>String(o.data||'')>=today).sort((a,b)=>String(a.data).localeCompare(String(b.data))).slice(0,6);
  const topUnits=topCounts(inYear,o=>o.unidade||'',6);
  const months=[];
  for(let i=0;i<12;i++){
    const d=new Date(Number(year),i,1);
    const key=`${year}-${String(i+1).padStart(2,'0')}`;
    const arr=all.filter(o=>String(o.data||'').slice(0,7)===key);
    const s=arr.filter(o=>o.tipo==='salao').length;
    const m=arr.filter(o=>o.tipo==='mudanca').length;
    months.push({
      key,
      label:d.toLocaleDateString('pt-BR',{month:'short'}).replace('.',''),
      total:arr.length,
      salaos:s,
      mudancas:m
    });
  }
  return {month,year,label:agendaReportPeriodLabel(month),inMonth,inYear,salaos,mudancas,salaosYear,mudancasYear,future,topUnits,months};
}

function csvDownload(filename, rows){
  const lines=(rows||[]).map(row=>row.map(v=>`"${String(v??'').replace(/"/g,'""')}"`).join(';'));
  const blob=new Blob(['\ufeff'+lines.join('\n')],{type:'text/csv;charset=utf-8'});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');
  a.href=url;
  a.download=filename;
  a.click();
  URL.revokeObjectURL(url);
}

function scheduleSmartLabel(iso){
  if(!iso)return {text:'Sem horário',cls:''};
  const when=new Date(iso);
  if(isNaN(when.getTime()))return {text:'Horário inválido',cls:'due'};
  const now=new Date();
  const sameDay=(a,b)=>a.getFullYear()===b.getFullYear()&&a.getMonth()===b.getMonth()&&a.getDate()===b.getDate();
  const tomorrow=new Date(now); tomorrow.setDate(now.getDate()+1);
  const hh=when.toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'});
  if(when.getTime()<=now.getTime())return {text:'Atrasado — será publicado ao abrir o app',cls:'due'};
  if(sameDay(when,now))return {text:'Publica hoje às '+hh,cls:'today'};
  if(sameDay(when,tomorrow))return {text:'Publica amanhã às '+hh,cls:'tomorrow'};
  return {text:'Publica em '+when.toLocaleDateString('pt-BR',{day:'2-digit',month:'2-digit'})+' às '+hh,cls:''};
}
