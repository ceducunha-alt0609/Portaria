  }

}

function crc32(str){
  let crc=0xFFFFFFFF;
  for(let i=0;i<str.length;i++){
    crc^=str.charCodeAt(i);
    for(let j=0;j<8;j++) crc=(crc>>>1)^(crc&1?0xEDB88320:0);
  }
  return((crc^0xFFFFFFFF)>>>0).toString(16).padStart(8,'0');
}

function buildBackupPayload(tipo='completo'){
  const now=new Date();
  let dados={};
  if(tipo==='completo')      dados={accesses:state.accesses,units:state.units,settings:state.settings,operational:state.operational,users:state.users.map(u=>({...u,senha:undefined})),logs:state.logs};
  else if(tipo==='moradores') dados={units:state.units};
  else if(tipo==='acessos')   dados={accesses:state.accesses};
  else if(tipo==='logs')      dados={logs:state.logs};

  const dadosStr=JSON.stringify(dados);
  return {
    _pp_backup:true,
    versao:'1.0',
    tipo,
    app:'Portaria Primavera',
    gerado_em:now.toISOString(),
    gerado_em_br:now.toLocaleString('pt-BR'),
    stats:{
      unidades:Object.keys(state.units||{}).length,
      acessos:(state.accesses||[]).length,
      logs:(state.logs||[]).length,
      usuarios:(state.users||[]).length,
    },
    checksum:crc32(dadosStr),
    dados
  };
}

/* Valida envelope de backup na importação */
function validateBackup(parsed){
  if(!parsed||typeof parsed!=='object') return {ok:false,msg:'Arquivo não é um JSON válido.'};
  // Formato antigo (state direto) — aceitar com aviso
  if(!parsed._pp_backup){
    if(parsed.accesses!==undefined||parsed.units!==undefined)
      return {ok:true,legado:true,msg:'Backup no formato antigo (sem metadados). Será importado com compatibilidade.'};
    return {ok:false,msg:'Arquivo não reconhecido como backup do Portaria Primavera.'};
  }
  // Formato novo — validar checksum
  const dadosStr=JSON.stringify(parsed.dados);
  const calc=crc32(dadosStr);
  if(calc!==parsed.checksum)
    return {ok:false,msg:`Checksum inválido — arquivo pode estar corrompido.\nEsperado: ${parsed.checksum}\nCalculado: ${calc}`};
  return {ok:true,legado:false,msg:''};
}

  const key=(document.getElementById('gdriveApiKeyConfig')?.value||'').trim();
  if(!cid || !cid.includes('.apps.googleusercontent.com')){
    showCloudToast?.('Informe um OAuth Client ID válido do Google','err',5000);
    return;
  }
  if(!key || !key.startsWith('AIza')){
    showCloudToast?.('Informe uma API Key válida do Google','err',5000);
    return;
  }
  localStorage.setItem('pp_gdrive_client_id',cid);
  localStorage.setItem('pp_gdrive_api_key',key);
  addLog?.('Integração Google Drive','Credenciais salvas em Central Administrativa → Integrações','sistema');
  updateIntegrationStatus();
  setDriveStatus('✅ Google Drive configurado. Clique em Conectar ao Google Drive.','ok');
  showCloudToast?.('Integração Google Drive salva ✅','ok',4000);
}

  const q=encodeURIComponent(`name='${GDRIVE_FOLDER}' and mimeType='application/vnd.google-apps.folder' and trashed=false`);

  });
  const folder=await cr.json();
  return folder.id;
}

function scheduleCloudAutoBackup(checkCatchUp=false){
  if(!cloudReady||!supabaseClient)return;
  clearTimeout(cloudTimer);
  const today=cloudTodayKey();
  const targetToday=cloudBackupTargetToday();
  const now=new Date();
  const last=localStorage.getItem(CLOUD_DAILY_KEY)||'';

  // Se o app abriu depois das 02:00 e ainda não fez backup hoje, faz um catch-up rápido.
  if(checkCatchUp && last!==today && now>=targetToday){
    setCloudStatus('☁️ Supabase conectado. Backup diário pendente: enviando em instantes...');
    cloudTimer=setTimeout(()=>cloudSaveDailyIfNeeded(),2500);
    return;
  }

  const next=cloudNextTarget();
  const delay=Math.max(1000,next.getTime()-Date.now());
  setCloudStatus(`☁️ Supabase conectado. Backup automático diário agendado para ${cloudTargetLabel(next)}.`);
  cloudTimer=setTimeout(()=>cloudSaveDailyIfNeeded(),delay);
}
async function cloudSaveDailyIfNeeded(){
  if(!cloudReady||!supabaseClient)return;
  const today=cloudTodayKey();
  if(localStorage.getItem(CLOUD_DAILY_KEY)===today){scheduleCloudAutoBackup(false);return;}
  const ok=await cloudSaveNow('auto-diario',true);
  if(ok)localStorage.setItem(CLOUD_DAILY_KEY,today);
  scheduleCloudAutoBackup(false);
}

function initCloud(){
  try{
    if(!window.supabase){setCloudStatus('☁️ Nuvem indisponível: biblioteca Supabase não carregou. O app segue local.');return false;}
    supabaseClient=window.supabase.createClient(SUPABASE_URL,SUPABASE_ANON_KEY);
    cloudReady=true;
    setTimeout(setupSupabaseRecoveryListener,0);
    scheduleCloudAutoBackup(true);
    startCloudSync();
    return true;
  }catch(e){console.warn('Falha Supabase',e);setCloudStatus('☁️ Falha ao conectar Supabase. O app segue local.');return false;}
}
function scheduleCloudAutoBackup(checkCatchUp=false){
  if(!cloudReady||!supabaseClient)return;
  clearTimeout(cloudTimer);
  const today=cloudTodayKey();
  const targetToday=cloudBackupTargetToday();
  const now=new Date();
  const last=localStorage.getItem(CLOUD_DAILY_KEY)||'';

  // Se o app abriu depois das 02:00 e ainda não fez backup hoje, faz um catch-up rápido.
  if(checkCatchUp && last!==today && now>=targetToday){
    setCloudStatus('☁️ Supabase conectado. Backup diário pendente: enviando em instantes...');
    cloudTimer=setTimeout(()=>cloudSaveDailyIfNeeded(),2500);
    return;
  }

  const next=cloudNextTarget();
  const delay=Math.max(1000,next.getTime()-Date.now());
  setCloudStatus(`☁️ Supabase conectado. Backup automático diário agendado para ${cloudTargetLabel(next)}.`);
  cloudTimer=setTimeout(()=>cloudSaveDailyIfNeeded(),delay);
}
async function cloudSaveDailyIfNeeded(){
  if(!cloudReady||!supabaseClient)return;
  const today=cloudTodayKey();
  if(localStorage.getItem(CLOUD_DAILY_KEY)===today){scheduleCloudAutoBackup(false);return;}
  const ok=await cloudSaveNow('auto-diario',true);
  if(ok)localStorage.setItem(CLOUD_DAILY_KEY,today);
  scheduleCloudAutoBackup(false);
}

async function handleUserRecoveryAuth(){
  if(!isAdmin())return;
  const id=val('uEditId');
  const u=(state.users||[]).find(x=>x.id===id); if(!u)return;
  const email=(val('uEmail')||u.email||'').trim().toLowerCase();
  if(!email){openAppConfirm({title:'E-mail necessário',message:'Informe e salve um e-mail de recuperação antes de ativar.',okText:'OK'});return;}
  if(email!==String(u.email||'').trim().toLowerCase()){
    openAppConfirm({title:'Salve primeiro',message:'O e-mail foi alterado. Salve o usuário antes de ativar a recuperação.',okText:'OK'});return;
  }
  if(!String(u.senha||'')){
    openAppConfirm({title:'Senha indisponível',message:'Defina uma senha local para este usuário antes de ativar a recuperação.',okText:'OK'});return;
  }
  if(u.authRecoveryPending===true||u.authRecoveryEnabled===true){
    await verifyUserRecoveryAuth(u); return;
  }
  openAppConfirm({
    title:'Ativar recuperação por e-mail',
    message:'O Portaria enviará ao Supabase o e-mail '+maskRecoveryEmail(email)+' e a senha atual deste usuário somente para criar o vínculo de autenticação. O login local continuará funcionando normalmente.',
    okText:'Ativar',cancelText:'Cancelar',
    onConfirm:async()=>{
      try{
        if(!cloudReady&&!initCloud())throw new Error('Supabase indisponível');
        const redirectTo=APP_PUBLIC_URL;
        const {data,error}=await supabaseClient.auth.signUp({email,password:String(u.senha),options:{emailRedirectTo:redirectTo}});
        if(error)throw error;
        try{await supabaseClient.auth.signOut()}catch(e){}
        u.authRecoveryPending=true;u.authRecoveryEnabled=false;u.authRecoveryEmail=email;
        addLog('Ativou recuperação por e-mail',u.nome+' • confirmação pendente','usuario');
        save();renderUserRecoveryAuth(u);
        openAppConfirm({title:'Confirme seu e-mail',message:'Solicitação enviada. Abra a mensagem recebida em '+maskRecoveryEmail(email)+' e confirme o cadastro. Depois volte aqui e clique em “Verificar ativação”.',okText:'Entendi'});
      }catch(e){
        console.warn('Falha ao ativar recuperação',e);
        openAppConfirm({title:'Não foi possível ativar',message:'O Supabase não concluiu a ativação. O login atual não foi alterado. Detalhe: '+(e?.message||'falha de autenticação'),okText:'OK'});
      }
    }
  });
}
async function verifyUserRecoveryAuth(u){
  try{
    if(!cloudReady&&!initCloud())throw new Error('Supabase indisponível');
    const email=String(u.email||'').trim().toLowerCase();
    const {data,error}=await supabaseClient.auth.signInWithPassword({email,password:String(u.senha||'')});
    if(error)throw error;
    try{await supabaseClient.auth.signOut()}catch(e){}
    u.authRecoveryPending=false;u.authRecoveryEnabled=true;u.authRecoveryVerifiedAt=new Date().toISOString();
    addLog('Verificou recuperação por e-mail',u.nome+' • '+maskRecoveryEmail(email),'usuario');
    save();renderUserRecoveryAuth(u);
    openAppConfirm({title:'Recuperação ativada',message:'✓ O e-mail foi validado. A opção “Esqueci minha senha” já pode solicitar uma redefinição segura para '+maskRecoveryEmail(email)+'.',okText:'OK'});
  }catch(e){
    console.warn('Verificação recuperação',e);
    const msg=String(e?.message||'');
    const pending=/confirm|confirmed|verification/i.test(msg);
    openAppConfirm({title:pending?'Confirmação ainda pendente':'Não foi possível verificar',message:pending?'Confirme primeiro a mensagem recebida no e-mail e tente novamente.':'A conta ainda não pôde ser validada. O login local continua normal. Detalhe: '+(msg||'falha de autenticação'),okText:'OK'});
  }
}

    document.getElementById('uSenha').value='';
    document.getElementById('uPerfil').value=u.perfil||'portaria';
    const turnoEl=document.getElementById('uTurno'); if(turnoEl)turnoEl.value=u.turno||'';
    document.getElementById('uAtivo').value=(u.ativo===false?'false':'true');
    if(title)title.textContent='Editar usuário';
    renderUserRecoveryAuth(u);
  }else{
    if(title)title.textContent='Novo usuário';
  }
  if(!id){const rb=document.getElementById('uRecoveryAuthBox');if(rb)rb.style.display='none';}
  if(modal)modal.classList.add('show');
}

function maskCPF(value){
  const d=digitsOnly(value).slice(0,11);
  if(d.length<=3)return d;

function getCommMenuById(id){
  return Array.from(document.querySelectorAll('.commMenu')).find(m=>String(m.dataset.id)===String(id));
}
function resetCommMenuPosition(menu){
  if(!menu)return;
  menu.style.removeProperty('position');
  menu.style.removeProperty('left');
  menu.style.removeProperty('right');
  menu.style.removeProperty('top');
  menu.style.removeProperty('bottom');
}
function positionCommMenu(menu,btn){
  if(!menu||!btn)return;
  const width=170;
  const rect=btn.getBoundingClientRect();
  const height=menu.offsetHeight||158;
  let left=rect.right-width+6;
  left=Math.max(10,Math.min(left,window.innerWidth-width-10));
  let top=rect.bottom-height-4;
  if(top<10) top=rect.bottom+8;
  menu.style.setProperty('position','fixed','important');
  menu.style.setProperty('left',left+'px','important');
  menu.style.setProperty('right','auto','important');
  menu.style.setProperty('top',top+'px','important');
  menu.style.setProperty('bottom','auto','important');
}
function toggleCommCardMenu(id,btn){
  const menu=getCommMenuById(id);
  if(!menu)return;
  const willOpen=!menu.classList.contains('open');
  document.querySelectorAll('.commMenu.open').forEach(m=>{m.classList.remove('open'); resetCommMenuPosition(m);});
  if(willOpen){
    menu.classList.add('open');
    positionCommMenu(menu,btn);
  }
}
function closeAllCommMenus(){
  document.querySelectorAll('.commMenu.open').forEach(m=>{m.classList.remove('open'); resetCommMenuPosition(m);});
}

}


function formatDateTimeBR(iso){
  if(!iso)return '—';
  try{return new Date(iso).toLocaleString('pt-BR',{day:'2-digit',month:'2-digit',year:'2-digit',hour:'2-digit',minute:'2-digit'});}
  catch(e){return '—';}
}
