function exportResidentsCsv(){

  Object.values(state.units||{}).forEach(u=>{
    const people=(u.people||[]).filter(p=>p&&p.nome);
    const vehicles=(u.vehicles||[]).filter(v=>v&&(v.placa||v.cor||v.modelo||v.vaga));
    if(!people.length&&!vehicles.length){
      rows.push([u.bloco||'',u.apto||'','','','','','','','','','']);
      return;
    }
    const max=Math.max(people.length,vehicles.length,1);
    for(let i=0;i<max;i++){
      const p=people[i]||{};
      const v=vehicles[i]||{};
      rows.push([u.bloco||'',u.apto||'',p.nome||'',p.tel||p.telefone||'',p.vinculo||'',p.app||'',p.bio||'',v.placa||'',v.cor||'',v.modelo||'',v.vaga||'']);
    }
  });
  csvDownload('relatorio-moradores-portaria-primavera.csv',rows);
  addLog?.('Exportou relatório de moradores em CSV',`${rows.length-1} linha(s)`,'relatorios');
  persistNoRender?.();
}

function exportResidentsCsv(){
  const rows=[['Bloco','Apto','Morador','Telefone','Vínculo','Usa app','Biometria','Placa','Cor','Modelo','Vaga']];
  Object.values(state.units||{}).forEach(u=>{
    const people=(u.people||[]).filter(p=>p&&p.nome);
    const vehicles=(u.vehicles||[]).filter(v=>v&&(v.placa||v.cor||v.modelo||v.vaga));
    if(!people.length&&!vehicles.length){
      rows.push([u.bloco||'',u.apto||'','','','','','','','','','']);
      return;
    }
    const max=Math.max(people.length,vehicles.length,1);
    for(let i=0;i<max;i++){
      const p=people[i]||{};
      const v=vehicles[i]||{};
      rows.push([u.bloco||'',u.apto||'',p.nome||'',p.tel||p.telefone||'',p.vinculo||'',p.app||'',p.bio||'',v.placa||'',v.cor||'',v.modelo||'',v.vaga||'']);
    }
  });
  csvDownload('relatorio-moradores-portaria-primavera.csv',rows);
  addLog?.('Exportou relatório de moradores em CSV',`${rows.length-1} linha(s)`,'relatorios');
  persistNoRender?.();
}

function csvDownload(filename, rows){
  const lines=(rows||[]).map(row=>row.map(v=>`"${String(v??'').replace(/"/g,'""')}"`).join(';'));
  const blob=new Blob(['\ufeff'+lines.join('\n')],{type:'text/csv;charset=utf-8'});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');
  a.href=url;
  a.download=filename;
  a.click();
  URL.revokeObjectURL(url);
}
