function crc32(str){
  let crc=0xFFFFFFFF;
  for(let i=0;i<str.length;i++){
    crc^=str.charCodeAt(i);
    for(let j=0;j<8;j++) crc=(crc>>>1)^(crc&1?0xEDB88320:0);
  }
  return((crc^0xFFFFFFFF)>>>0).toString(16).padStart(8,'0');
}

function validateBackup(parsed){
  if(!parsed||typeof parsed!=='object') return {ok:false,msg:'Arquivo não é um JSON válido.'};
  // Formato antigo (state direto) — aceitar com aviso
  if(!parsed._pp_backup){
    if(parsed.accesses!==undefined||parsed.units!==undefined)
      return {ok:true,legado:true,msg:'Backup no formato antigo (sem metadados). Será importado com compatibilidade.'};
    return {ok:false,msg:'Arquivo não reconhecido como backup do Portaria Primavera.'};
  }
  // Formato novo — validar checksum
  const dadosStr=JSON.stringify(parsed.dados);
  const calc=crc32(dadosStr);
  if(calc!==parsed.checksum)
    return {ok:false,msg:`Checksum inválido — arquivo pode estar corrompido.\nEsperado: ${parsed.checksum}\nCalculado: ${calc}`};
  return {ok:true,legado:false,msg:''};
}

function applyBackupData(parsed,file){
  const status=document.getElementById('backupStatus');
  let dados, tipo;
  if(parsed._pp_backup){
    dados=parsed.dados; tipo=parsed.tipo||'completo';
  } else {
    dados=parsed; tipo='completo'; // formato legado
  }
  if(tipo==='completo'){
    state={...initial(),...dados};
  } else if(tipo==='moradores'){
    state.units={...dados.units||{}};
  } else if(tipo==='acessos'){
    state.accesses=[...dados.accesses||[]];
  } else if(tipo==='logs'){
    state.logs=[...dados.logs||[]];
  }
  if(!Array.isArray(state.logs))state.logs=[];
  if(!Array.isArray(state.users)||!state.users.length)state.users=defaultUsers();
  state.users.forEach(u=>normalizeUserOperational(u));
  ensureSettings(); normalizeImportedVehicles?.(); normalizeAccesses?.();
  localStorage.setItem(KEY,JSON.stringify(state));
  addLog('Restaurou backup','Arquivo: '+file.name+' (tipo: '+tipo+')','backup');
  applyTheme(); renderAll(); applyPermissions();
  const info=parsed._pp_backup?` — gerado em ${parsed.gerado_em_br}`:'';
  if(status)status.textContent=`✅ Backup restaurado: ${file.name}${info}`;
}

  const defaults=defaultUsers();
  const byId=new Map(defaults.map(u=>[u.id,u]));
  const byLogin=new Map(defaults.map(u=>[u.login,u]));
  return (Array.isArray(users)?users:[]).map(u=>{

  });
}
